def refresh_dashboard(dashboard_id: str) -> RefreshResult:
    """Only refreshes a Power BI dashboard once every upstream check passes."""
    checks = run_precheck_suite(dashboard_id)   # data-freshness / row-count / null-rate checks

    if not all(c.passed for c in checks):
        failing = [c.name for c in checks if not c.passed]
        log_status(dashboard_id, "BLOCKED", reason=failing)
        return RefreshResult(status="blocked", failing_checks=failing)

    result = trigger_powerbi_refresh(dashboard_id)
    log_status(dashboard_id, result.status)
    post_slack_status_card(dashboard_id, result)   # renders the live PNG status board

    if result.status == "failed" and is_repeat_failure(dashboard_id):
        file_report_task(dashboard_id, result.error)  # auto-track recurring failures
    return result
